def compute_order_amount(history_demand, history_inventory, history_lost, holding_cost, lost_sales_cost):
    if not history_demand:
        return 0
    
    # Algorithm parameters
    demand_window = 3
    momentum_factor = 0.4
    quantile_level = 0.75
    cost_sensitivity = 1.5
    
    # Momentum-adjusted demand forecast
    if len(history_demand) >= demand_window:
        recent_trend = history_demand[-1] - history_demand[-demand_window]
        base_forecast = sum(history_demand[-demand_window:]) / demand_window
        forecast = base_forecast + momentum_factor * recent_trend
    else:
        forecast = sum(history_demand) / len(history_demand)
    
    # Quantile-based safety stock
    if len(history_demand) > 1:
        sorted_demand = sorted(history_demand)
        idx = min(int(quantile_level * len(sorted_demand)), len(sorted_demand)-1)
        safety_stock = max(0, sorted_demand[idx] - forecast)
    else:
        safety_stock = 0
    
    # Cost-asymmetric adjustment
    cost_ratio = (lost_sales_cost**cost_sensitivity) / (holding_cost + lost_sales_cost**cost_sensitivity)
    adjusted_forecast = forecast * (1 + cost_ratio)
    
    # Current inventory position
    inventory_position = history_inventory[-1] if history_inventory else 0
    
    # Order amount calculation
    order_amount = max(0, int(adjusted_forecast + safety_stock - inventory_position))
    
    return order_amount

def compute_order_amount(history_demand, history_inventory, history_lost, holding_cost, lost_sales_cost):
    if not history_demand:
        return 0
    
    # Cluster demand into high/medium/low bins
    demand_sorted = sorted(history_demand)
    n = len(demand_sorted)
    low_demand = demand_sorted[:n//3]
    medium_demand = demand_sorted[n//3:2*n//3]
    high_demand = demand_sorted[2*n//3:]
    
    # Calculate cluster probabilities
    p_low = len(low_demand) / n
    p_med = len(medium_demand) / n
    p_high = len(high_demand) / n
    
    # Expected demand per cluster
    avg_low = sum(low_demand) / (len(low_demand) or 1)
    avg_med = sum(medium_demand) / (len(medium_demand) or 1)
    avg_high = sum(high_demand) / (len(high_demand) or 1)
    
    # Weighted expected demand
    expected_demand = p_low*avg_low + p_med*avg_med + p_high*avg_high
    
    # Lost sales probability adjustment
    lost_prob = sum(1 for x in history_lost if x > 0) / (len(history_lost) or 1)
    adjusted_demand = expected_demand * (1 + lost_prob)
    
    # Cost-sensitive quantile
    cost_ratio = lost_sales_cost / (holding_cost + lost_sales_cost)
    quantile_idx = min(int(cost_ratio * n), n-1)
    safety_stock = demand_sorted[quantile_idx]
    
    # Current inventory
    current_inventory = history_inventory[-1] if history_inventory else 0
    
    # Order amount
    order_amount = max(0, adjusted_demand + safety_stock - current_inventory)
    
    return order_amount

    buffer_stock = forecast_demand * buffer_factor
    
    order_amount = max(0, weighted_demand + buffer_stock - current_inventory)
    
    return order_amount
