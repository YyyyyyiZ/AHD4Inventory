# Description: This algorithm implements a demand-adaptive threshold policy that dynamically adjusts order quantities based on recent demand patterns and cost ratios, using exponential smoothing for demand forecasting and a safety factor tuned to cost parameters.
def compute_order_amount(history_demand, history_inventory, history_lost, holding_cost, lost_sales_cost):
    alpha = 0.1  # Optimized
    safety_factor = 1.3527921117321604  # Optimized
    min_order = 5  # Optimized
    
    if len(history_demand) < 2:
        forecast = sum(history_demand) / len(history_demand) if history_demand else 0
    else:
        forecast = alpha * history_demand[-1] + (1 - alpha) * (sum(history_demand[:-1]) / (len(history_demand) - 1))
    
    cost_ratio = lost_sales_cost / (holding_cost + lost_sales_cost)
    adjusted_forecast = forecast * (1 + safety_factor * cost_ratio)
    
    current_inv = history_inventory[-1]
    order_amount = max(adjusted_forecast - current_inv, min_order)
    
    return order_amount

# Description: This algorithm uses demand volatility-weighted ordering with asymmetric cost response, where order quantities are adjusted based on normalized demand fluctuations and a non-linear cost sensitivity function, incorporating lead-time anticipation through exponential smoothing.
def compute_order_amount(history_demand, history_inventory, history_lost, holding_cost, lost_sales_cost):
    base_order = 169.99999997269938  # Optimized
    volatility_weight = 1.0  # Optimized
    cost_sensitivity = 0.5  # Optimized
    lead_time_factor = 0.5  # Optimized
    smoothing_alpha = 0.9  # Optimized
    
    if not history_demand:
        return base_order * lead_time_factor
    
    current_inventory = history_inventory[-1] if history_inventory else 0
    
    # Demand volatility calculation
    mean_demand = sum(history_demand) / len(history_demand)
    demand_variance = sum((d - mean_demand)**2 for d in history_demand) / len(history_demand)
    normalized_volatility = min(1.0, (demand_variance**0.5) / (mean_demand + 1e-6))
    
    # Asymmetric cost response
    cost_ratio = (lost_sales_cost / (holding_cost + lost_sales_cost)) ** cost_sensitivity
    volatility_adjustment = 1 + (normalized_volatility * volatility_weight * cost_ratio)
    
    # Exponential smoothing with lead time anticipation
    last_demand = history_demand[-1]
    smoothed_demand = smoothing_alpha * last_demand + (1 - smoothing_alpha) * mean_demand
    adjusted_demand = smoothed_demand * volatility_adjustment * lead_time_factor
    
    # Order calculation
    target = max(base_order, adjusted_demand)
    order_amount = max(0, target - current_inventory)
    
    return order_amount

# Description: This algorithm uses a hybrid approach combining quantile-based demand forecasting with dynamic safety stock calculation, where orders are determined by a demand percentile adjusted by a non-linear cost-sensitive safety factor and lead time consideration.
def compute_order_amount(history_demand, history_inventory, history_lost, holding_cost, lost_sales_cost):
    demand_percentile = 0.75  # Optimized
    safety_factor = 0.8099099373712129  # Optimized
    cost_exponent = 0.10919396434999616  # Optimized
    min_order = 10.0  # Optimized
    decay_rate = 0.01  # Optimized
    
    if not history_demand:
        return min_order
    
    # Calculate demand percentile
    sorted_demand = sorted(history_demand)
    idx = min(len(sorted_demand)-1, int(len(sorted_demand) * demand_percentile))
    percentile_demand = sorted_demand[idx]
    
    # Calculate cost-sensitive adjustment
    cost_ratio = lost_sales_cost / max(holding_cost, 0.001)
    cost_adjustment = (cost_ratio ** cost_exponent) * safety_factor
    
    # Recent demand weighting
    weighted_demand = 0.0
    total_weight = 0.0
    for i, d in enumerate(reversed(history_demand)):
        weight = (1 - decay_rate) ** i
        weighted_demand += d * weight
        total_weight += weight
    recent_demand = weighted_demand / total_weight if total_weight > 0 else percentile_demand
    
    # Combined forecast with safety stock
    base_demand = max(percentile_demand, recent_demand)
    safety_stock = base_demand * cost_adjustment
    
    current_inv = history_inventory[-1] if history_inventory else 0
    order_amount = max(min_order, (base_demand + safety_stock) - current_inv)
    
    return order_amount

# Description: This algorithm uses a demand volatility-adjusted approach with dynamic safety stock calculation, incorporating weighted recent demand deviations and cost-ratio-dependent inventory targets.
def compute_order_amount(history_demand, history_inventory, history_lost, holding_cost, lost_sales_cost):
    demand_volatility_weight = 0.1  # Optimized
    recent_window = 3  # Optimized
    cost_ratio_power = 1.9958732326932997  # Optimized
    min_order = 50.0  # Optimized
    safety_multiplier = 2.029592498404381  # Optimized
    
    if not history_demand:
        return min_order
    
    current_inv = history_inventory[-1] if history_inventory else 0
    avg_demand = sum(history_demand) / len(history_demand)
    
    recent_demand = history_demand[-recent_window:] if len(history_demand) > recent_window else history_demand
    demand_deviation = sum(abs(d - avg_demand) for d in recent_demand) / len(recent_demand)
    volatility_adjusted = avg_demand + (demand_volatility_weight * demand_deviation)
    
    cost_ratio = lost_sales_cost / (holding_cost + lost_sales_cost)
    cost_adjusted = volatility_adjusted * (cost_ratio ** cost_ratio_power)
    
    safety_stock = safety_multiplier * cost_adjusted
    target_inv = cost_adjusted + safety_stock
    
    order_amount = max(min_order, target_inv - current_inv)
    return order_amount

# Description: This algorithm uses a demand pattern clustering approach with cost-sensitive reorder points, dynamically adjusting order quantities based on clustered demand behavior and real-time inventory position.
def compute_order_amount(history_demand, history_inventory, history_lost, holding_cost, lost_sales_cost):
    cluster_window = 5  # Optimized
    reorder_multiplier = 2.0375350511323678  # Optimized
    min_order = 71.94537242368094  # Optimized
    cost_sensitivity = 0.12003003225549509  # Optimized
    
    if not history_demand:
        return min_order
    
    current_inv = history_inventory[-1] if history_inventory else 0
    cost_ratio = lost_sales_cost / (holding_cost + lost_sales_cost)
    
    # Cluster demand patterns
    if len(history_demand) >= cluster_window:
        clusters = []
        for i in range(len(history_demand) - cluster_window + 1):
            cluster = history_demand[i:i+cluster_window]
            clusters.append((sum(cluster)/len(cluster), max(cluster)-min(cluster)))
        avg_cluster = sum(c[0] for c in clusters) / len(clusters)
        volatility = sum(c[1] for c in clusters) / len(clusters)
    else:
        avg_cluster = sum(history_demand) / len(history_demand)
        volatility = max(history_demand) - min(history_demand)
    
    # Cost-sensitive adjustment
    adjusted_demand = avg_cluster + (volatility * cost_ratio * cost_sensitivity)
    reorder_point = adjusted_demand * reorder_multiplier
    
    order_amount = max(min_order, reorder_point - current_inv)
    return order_amount

# Description: This algorithm uses demand clustering with cost-weighted buffer zones and lead-time-aware inventory positioning, dynamically adjusting order quantities based on demand pattern similarity and real-time cost ratios.
def compute_order_amount(history_demand, history_inventory, history_lost, holding_cost, lost_sales_cost):
    cluster_threshold = 0.35  # Optimized
    base_zone = 44.98786270278207  # Optimized
    cost_exponent = 0.22298479405808364  # Optimized
    lead_time_penalty = 1.0208406375706451  # Optimized
    recent_weight = 0.1  # Optimized

    if not history_demand:
        return base_zone * 2

    current_inventory = history_inventory[-1] if history_inventory else 0
    
    # Demand clustering
    recent_demand = history_demand[-3:] if len(history_demand) >= 3 else history_demand
    avg_recent = sum(recent_demand) / len(recent_demand) if recent_demand else 0
    cluster_matches = [d for d in history_demand if abs(d - avg_recent) / (avg_recent + 1e-6) <= cluster_threshold]
    
    # Cluster-based forecast
    if cluster_matches:
        forecast = sum(cluster_matches) / len(cluster_matches)
    else:
        forecast = avg_recent
    
    # Cost-weighted buffer zone
    cost_ratio = (lost_sales_cost / (holding_cost + lost_sales_cost)) ** cost_exponent
    buffer_zone = base_zone * (1 + cost_ratio)
    
    # Lead-time adjusted target with recent weighting
    historical_avg = sum(history_demand) / len(history_demand)
    blended_forecast = recent_weight * forecast + (1 - recent_weight) * historical_avg
    target = (blended_forecast * lead_time_penalty) + buffer_zone
    
    order_amount = max(0, target - current_inventory)
    
    return order_amount

# Description: This algorithm uses a hybrid of moving window demand forecasting and Markov chain-based state transitions between inventory levels, with cost-proportional dynamic buffer sizing and lead-time-aware order smoothing.
def compute_order_amount(history_demand, history_inventory, history_lost, holding_cost, lost_sales_cost):
    window_size = 4  # Optimized
    state_transition_weight = 0.23939040985144583  # Optimized
    cost_buffer_multiplier = 0.36436089990891957  # Optimized
    lead_time_smoothing = 0.1  # Optimized
    base_buffer = 57.06838715300055  # Optimized
    
    if not history_demand:
        return base_buffer * 1.5
    
    current_inventory = history_inventory[-1] if history_inventory else 0
    
    # Moving window demand forecast
    window = history_demand[-window_size:] if len(history_demand) >= window_size else history_demand
    window_avg = sum(window) / len(window) if window else 0
    
    # State transition probability adjustment
    if len(history_inventory) >= 2:
        transitions = [history_inventory[i+1] - history_inventory[i] for i in range(len(history_inventory)-1)]
        transition_avg = sum(transitions) / len(transitions)
        state_adjustment = transition_avg * state_transition_weight
    else:
        state_adjustment = 0
    
    # Cost-proportional dynamic buffer
    cost_ratio = lost_sales_cost / (holding_cost + 1e-6)
    dynamic_buffer = base_buffer * (1 + cost_ratio * cost_buffer_multiplier)
    
    # Lead-time aware smoothing
    forecast = (window_avg + state_adjustment) * lead_time_smoothing
    
    target = forecast + dynamic_buffer
    order_amount = max(0, target - current_inventory)
    
    return order_amount
