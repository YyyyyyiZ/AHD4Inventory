# Description: An adaptive threshold policy that dynamically adjusts order-up-to levels based on demand volatility and cost-optimal service levels, incorporating lead time through a rolling horizon approach.
def compute_order_amount(history_demand, history_inventory, history_lost, holding_cost, lost_sales_cost):
    if not history_demand:
        return 100
    
    current_inv = history_inventory[-1] if history_inventory else 0
    cost_ratio = lost_sales_cost / (holding_cost + lost_sales_cost)
    
    # Calculate demand volatility
    mean_demand = sum(history_demand) / len(history_demand)
    variance = sum((d - mean_demand)**2 for d in history_demand) / len(history_demand)
    std_dev = variance**0.5
    volatility = std_dev / mean_demand if mean_demand > 0 else 0
    
    # Dynamic adjustment factor
    adjustment = 1 + (volatility * cost_ratio)
    base_order = mean_demand * adjustment
    
    # Lead time compensation
    lead_time_demand = mean_demand * 1  # lead time = 1 period
    
    target_level = base_order + lead_time_demand
    order_amount = max(0, target_level - current_inv)
    
    return order_amount

import numpy as np

def compute_order_amount(history_demand, history_inventory, history_lost, holding_cost, lost_sales_cost):
    if not history_demand:
        return 100
    
    # Initialize prediction models
    models = [
        lambda h: np.mean(h[-3:]) if len(h) >= 3 else np.mean(h) if h else 0,  # Moving average
        lambda h: h[-1] if h else 0,  # Last period
        lambda h: np.median(h[-5:]) if len(h) >= 5 else np.median(h) if h else 0  # Median
    ]
    
    # Track model performance
    if not hasattr(compute_order_amount, 'model_weights'):
        compute_order_amount.model_weights = [1.0] * len(models)
    
    # Update weights based on recent performance
    if len(history_demand) > 1:
        errors = []
        for model in models:
            pred = model(history_demand[:-1])
            errors.append(abs(pred - history_demand[-1]))
        min_error = min(errors)
        for i in range(len(models)):
            compute_order_amount.model_weights[i] *= 0.9 + 0.1 * (errors[i] == min_error)
    
    # Select model probabilistically
    weights = np.array(compute_order_amount.model_weights)
    probs = weights / weights.sum()
    chosen_model = np.random.choice(models, p=probs)
    
    # Predict demand
    predicted_demand = chosen_model(history_demand)
    
    # Cost-aware optimization
    critical_ratio = lost_sales_cost / (holding_cost + lost_sales_cost)
    safety_stock = np.percentile(history_demand, min(100, 100 * critical_ratio * 1.2)) if history_demand else 0
    
    target_level = predicted_demand + safety_stock
    order_amount = max(0, target_level - history_inventory[-1])
    
    return order_amount
