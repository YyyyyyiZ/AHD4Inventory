# Description: This algorithm uses a moving average demand forecast with volatility adjustment, incorporates cost-sensitive safety stock, and applies a non-linear response to inventory gaps.
def compute_order_amount(history_demand, history_inventory, history_lost, holding_cost, lost_sales_cost):
    window_size = 3  # Optimized
    safety_factor = 3.0  # Optimized
    response_power = 0.7401240334624253  # Optimized
    cost_sensitivity = 1.0  # Optimized
    
    if len(history_demand) == 0:
        return 0
    
    # Moving average forecast with volatility adjustment
    if len(history_demand) >= window_size:
        forecast = sum(history_demand[-window_size:]) / window_size
        volatility = max(history_demand[-window_size:]) - min(history_demand[-window_size:])
    else:
        forecast = sum(history_demand) / len(history_demand)
        volatility = 0
    
    # Cost-adjusted safety stock
    cost_ratio = holding_cost / (holding_cost + lost_sales_cost)
    adjusted_safety = safety_factor * forecast * (1 + cost_sensitivity * (1 - cost_ratio)) + volatility
    
    # Non-linear response to inventory gap
    inventory_gap = adjusted_safety - history_inventory[-1]
    if inventory_gap > 0:
        order_amount = inventory_gap ** response_power
    else:
        order_amount = 0
    
    return order_amount

def compute_order_amount(history_demand, history_inventory, history_lost, holding_cost, lost_sales_cost):
    initial_order = 100  # Optimized
    min_history = 5  # Optimized
    risk_threshold = 0.75  # Optimized
    buffer_multiplier = 1.0371652215636478  # Optimized
    service_sensitivity = 0.5  # Optimized
    
    if len(history_demand) < min_history:
        return initial_order
    
    # Calculate recent service level
    recent_periods = min(5, len(history_lost))
    service_level = 1 - (sum(history_lost[-recent_periods:]) / (sum(history_demand[-recent_periods:]) + 1e-6))
    
    # Dynamic risk adjustment based on costs and service
    cost_ratio = lost_sales_cost / (holding_cost + 1e-6)
    adjusted_risk = risk_threshold * (1 + service_sensitivity * (1 - service_level)) * (cost_ratio / (1 + cost_ratio))
    
    # Quantile-based demand forecast
    sorted_demand = sorted(history_demand)
    quantile_idx = min(len(sorted_demand)-1, int(adjusted_risk * len(sorted_demand)))
    forecast = sorted_demand[quantile_idx]
    
    # Lead time buffer
    buffer = forecast * buffer_multiplier * (1 + (1 - service_level))
    
    current_inventory = history_inventory[-1]
    order_amount = max(0, (forecast + buffer) - current_inventory)
    
    return order_amount

    target_level = weighted_demand + momentum + buffer_stock
    
    # Order calculation
