import math

def compute_order_amount(current_inventory, pipeline_inventory, history_demand, holding_cost, lost_sales_cost, lead_time):
    if not history_demand:
        return 0
    
    # Exponential smoothing parameters
    alpha = 0.3  # Smoothing factor for demand
    beta = 0.2   # Smoothing factor for demand variability
    
    # Initialize forecast and variability
    forecast = history_demand[0]
    variability = 0
    
    # Exponential smoothing for demand and variability
    for d in history_demand:
        forecast = alpha * d + (1 - alpha) * forecast
        variability = beta * abs(d - forecast) + (1 - beta) * variability
    
    # Cost ratio adjustment
    cost_ratio = lost_sales_cost / (holding_cost + lost_sales_cost)
    
    # Dynamic safety stock with lead-time adjustment
    safety_stock = math.sqrt(lead_time) * variability * (1 + cost_ratio)
    
    # Demand surge detection (recent 3-period average vs historical)
    recent_window = min(3, len(history_demand))
    recent_avg = sum(history_demand[-recent_window:]) / recent_window
    surge_factor = max(1, recent_avg / forecast)
    
    # Target inventory level with surge adjustment
    target_level = forecast * (lead_time + 1) * surge_factor + safety_stock
    
    # Calculate order amount
    incoming_orders = sum(pipeline_inventory)
    inventory_position = current_inventory + incoming_orders
    order_amount = max(0, target_level - inventory_position)

        return max(0, 100 - current_inventory - sum(pipeline_inventory))

    sigma_prior = max(np.std(history_demand), 1.0)

    recent_window = min(3, len(history_demand))
    recent_demand = history_demand[-recent_window:]
    mu_likelihood = np.mean(recent_demand)
    sigma_likelihood = max(np.std(recent_demand), 1.0)
    
    # Bayesian posterior update
    precision_prior = 1/(sigma_prior**2)
    precision_likelihood = 1/(sigma_likelihood**2)
    mu_posterior = (precision_prior*mu_prior + precision_likelihood*mu_likelihood)/(precision_prior + precision_likelihood)
    sigma_posterior = np.sqrt(1/(precision_prior + precision_likelihood))
    
    # Cost-sensitive quantile for service level
    critical_ratio = lost_sales_cost / (lost_sales_cost + holding_cost)
    z_score = norm.ppf(critical_ratio)
    
    # Lead-time adjusted forecast

    lead_time_std = sigma_posterior * np.sqrt(lead_time + 1)
    safety_stock = z_score * lead_time_std

    order_amount = max(0, lead_time_demand + safety_stock - net_inventory)
