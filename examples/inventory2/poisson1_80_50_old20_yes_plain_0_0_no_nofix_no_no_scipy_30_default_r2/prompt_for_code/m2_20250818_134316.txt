def compute_order_amount(current_inventory, pipeline_inventory, history_demand, holding_cost, lost_sales_cost, lead_time):
    window_size = 5  # Optimized
    base_safety_stock = 10  # Optimized
    cost_ratio_factor = 0.5  # Optimized
    
    if len(history_demand) >= window_size:
        forecast = sum(history_demand[-window_size:]) / window_size
        demand_std = (sum((d - forecast)**2 for d in history_demand[-window_size:]) / window_size)**0.5
    else:
        forecast = 80.04
        demand_std = 8.82
    
    cost_ratio = lost_sales_cost / (holding_cost + lost_sales_cost)
    safety_stock = base_safety_stock + cost_ratio_factor * cost_ratio * demand_std
    
    target_inventory = forecast * (lead_time + 1) + safety_stock
    order_amount = max(0, int(target_inventory - current_inventory - sum(pipeline_inventory)))
    
    return order_amount
