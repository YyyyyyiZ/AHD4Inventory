def compute_order_amount(current_inventory, pipeline_inventory, history_demand, holding_cost, lost_sales_cost, lead_time):
    demand_window = 15  # Optimized
    volatility_window = 7  # Optimized
    base_smoothing = 0.3  # Optimized
    cost_sensitivity = 0.85  # Optimized
    min_order = 10  # Optimized
    max_order = 120  # Optimized
    
    if not history_demand:
        demand_mean = 80.04
        demand_std = 8.82
    else:
        recent_demand = history_demand[-demand_window:] if len(history_demand) >= demand_window else history_demand
        demand_mean = sum(recent_demand) / len(recent_demand)
        
        volatility_demand = history_demand[-volatility_window:] if len(history_demand) >= volatility_window else history_demand
        demand_std = (sum((d - demand_mean)**2 for d in volatility_demand) / len(volatility_demand))**0.5 if len(volatility_demand) > 1 else 8.82
    
    critical_ratio = lost_sales_cost / (holding_cost + lost_sales_cost)
    z_score = cost_sensitivity * (2 * critical_ratio - 1)
    
    safety_stock = z_score * demand_std * (lead_time + 1)**0.5
    reorder_point = demand_mean * (lead_time + 1) + safety_stock
    
    incoming = sum(pipeline_inventory)
    inventory_position = current_inventory + incoming
    
    if inventory_position < reorder_point:
        order_amount = reorder_point - inventory_position
        order_amount = base_smoothing * order_amount + (1 - base_smoothing) * demand_mean
    else:
        order_amount = demand_mean * (1 - critical_ratio)
    
    order_amount = max(min_order, min(max_order, round(order_amount)))
    
    return order_amount
